82 struct FmTwrSegment
83 {
84  double H;
85  double D;
86  int N;
87  FmTwrSegment(double h = 10.0, double d = 5.0, int n = 5) : H(h), D(d), N(n) {}
88  bool operator==(const FmTwrSegment& s) const
89  { return H == s.H && D == s.D && N == s.N; }
90 };

103 template<>
104 inline void FFaField<FmTwrSegmentVec>::write(std::ostream& os) const
105 {
106  for (size_t i = 0; i < myData.size(); i++)
107  {
108  // write up to 9 numbers on a row
109  if (i) os << (i%3 ? " " : "\n\t");
110  os << myData[i].H <<" "<< myData[i].D <<" "<< myData[i].N;
111  }
112 }
113 
114 template<>
115 inline void FFaField<FmTwrSegmentVec>::read(std::istream& is, FmTwrSegmentVec& v) const
116 {
117  v.clear();
118  double h, d, n;
119  while (is)
120  {
121  is >> h >> d >> n;
122  if (is)
123  v.push_back(FmTwrSegment(h,d,n));
124  }
125  is.clear();
126 }
