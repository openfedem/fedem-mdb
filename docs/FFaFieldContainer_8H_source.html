81  template<class T>
82  void getReferringObjs(std::vector<T*>& vecToFill,
83  const std::string& context = "",
84  bool sortOnId = false) const
85  {
86  T* ptr;
87  if (sortOnId)
88  {
89  // Return the objects in sorted order w.r.t. their user ID
90  ObjectVec allRefs;
91  this->getReferringObjs(allRefs,context,sortOnId);
92  for (FFaFieldContainer* ref : allRefs)
93  if ((ptr = dynamic_cast<T*>(ref)))
94  vecToFill.push_back(ptr);
95  }
96  else
97  {
98  // Return the objects in arbitrary order
99  // (might be different in two runs of the same problem).
100  FFaFieldContainer* ref = this->getNext(context,true);
101  for (ptr = NULL; ref; ref = this->getNext(context))
102  if ((ptr = dynamic_cast<T*>(ref)))
103  vecToFill.push_back(ptr);
104  }
105  }
